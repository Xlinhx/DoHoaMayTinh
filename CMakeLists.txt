cmake_minimum_required(VERSION 3.24)
project(claw_lab VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/Dependencies.cmake)
add_library(claw_core src/core.cpp src/assets/GltfLoader.cpp)
target_include_directories(claw_core PUBLIC src third_party/tinygltf)
target_link_libraries(claw_core PUBLIC glm::glm)
target_compile_definitions(claw_core PUBLIC GLM_ENABLE_EXPERIMENTAL)
if(MINGW)
  add_link_options(-static-libgcc -static-libstdc++)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  add_executable(claw_lab WIN32 src/main.cpp src/graphics/Renderer.cpp)
  target_link_libraries(claw_lab PRIVATE claw_core glfw glad imgui opengl32)
  add_custom_command(TARGET claw_lab POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:claw_lab>/assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/shaders $<TARGET_FILE_DIR:claw_lab>/shaders)
  install(TARGETS claw_lab RUNTIME DESTINATION .)
  install(DIRECTORY assets shaders DESTINATION .)
  install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
  install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md")
  install(DIRECTORY licenses/ DESTINATION licenses)
endif()
enable_testing()
add_executable(claw_tests tests/core_tests.cpp)
target_link_libraries(claw_tests PRIVATE claw_core Catch2::Catch2WithMain)
target_compile_definitions(claw_tests PRIVATE SOURCE_ROOT="${CMAKE_SOURCE_DIR}")
foreach(group resource transform camera grab round import)
  add_test(NAME ${group} COMMAND claw_tests "[${group}]")
endforeach()
add_executable(camera_floor_regression tests/camera_floor_render.cpp src/graphics/Renderer.cpp)
target_link_libraries(camera_floor_regression PRIVATE claw_core glfw glad opengl32)
target_compile_definitions(camera_floor_regression PRIVATE SOURCE_ROOT="${CMAKE_SOURCE_DIR}")
